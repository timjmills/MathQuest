// Critic r1 sweep: fact rows, 4 skills (+ div forms) x S/M/L x columns x seeds.
// Per cell: ink bbox (all descendants except label) vs cell box, label keep-out, min font size of digits.
const path = require('path');
const fs = require('fs');
const ROOT = '/home/user/MathQuest';
const { open } = require(path.join(ROOT, 'tests/lib/ws-harness.cjs'));
const OUT = process.argv[2];
const ONLY = process.argv[3] || '';
const SHOTS = process.argv.includes('--shots'); const SHOTONLY = (process.argv.find(a=>a.startsWith('--shot='))||'').slice(7).split(',').filter(Boolean);
fs.mkdirSync(OUT, { recursive: true });
const configs = [];
const SEEDS3=[11,4242,90001];
const single = [['addition','add_facts',{},[11]],['subtraction','sub_facts',{},[11]],['multiplication','mult_facts',{},[11]],
 ['division','div_facts',{divForm:'standard'},SEEDS3],['division','div_facts',{divForm:'vertical'},SEEDS3],['division','div_facts',{divForm:'fraction'},SEEDS3],
 ['division','div_facts',{divForm:'long'},SEEDS3],['division','div_facts',{divForm:'mix'},SEEDS3],['division','div_facts_long',{},SEEDS3]];
for (const [c,s,o,seeds] of single) for (const size of ['S','M','L']) for (const columns of ['auto',5,6,8,10]) for (const seed of seeds)
  configs.push({ tag: s+(o.divForm?'-'+o.divForm:''), skills:[{categoryId:c,skillId:s,opts:o}], size, columns, seed });
const A={categoryId:'addition',skillId:'add_facts',opts:{}}, Su={categoryId:'subtraction',skillId:'sub_facts',opts:{}}, Mu={categoryId:'multiplication',skillId:'mult_facts',opts:{}};
const D=(f)=>({categoryId:'division',skillId:'div_facts',opts:{divForm:f}});
const mixes=[['mix4-Mix',[A,Su,Mu,D('mix')]],['mix4-Long',[A,Su,Mu,D('long')]],['mix3-Frac',[A,Su,D('fraction')]],['mix2-Std',[Mu,D('standard')]],['mix4-Vert',[A,Su,Mu,D('vertical')]],['mix2-LongPreset',[A,{categoryId:'division',skillId:'div_facts_long',opts:{}}]]];
for (const [tag,skills] of mixes) for (const size of ['S','M','L']) for (const columns of ['auto',5,6,8,10]) for (const seed of [11,4242])
  configs.push({ tag, skills, size, columns, seed });
(async () => {
  const { page, close, problems } = await open();
  const rows = [];
  for (const cf of configs) {
    const id = `${cf.tag}-${cf.size}-c${cf.columns}-s${cf.seed}`;
    if (ONLY && !id.includes(ONLY)) continue;
    const built = await page.evaluate(async (cf) => {
      try {
        const r = await window.buildSheet({ role: 'fact-rows', sections: [{ skills: cf.skills, columns: cf.columns }], size: cf.size, look: 'auto', paper: 'A4', seed: cf.seed, form: 'A', key: true });
        return { pupil: window.sheetDocument(r.pupilHtml, 't', { paper: 'A4' }), key: window.sheetDocument(r.keyHtml, 't key', { paper: 'A4' }), fits: r.fits, notes: r.notes, pageCount: r.pageCount, keyPageCount: r.keyPageCount, n: r.items.length, ans: r.items.map(i => [i.text, String(typeof i.ans === 'object' ? JSON.stringify(i.ans) : i.ans)]) };
      } catch (e) { return { error: String(e && e.stack || e) }; }
    }, cf);
    if (built.error) { rows.push({ id, error: built.error }); continue; }
    const res = {};
    for (const which of ['pupil', 'key']) {
      const sheet = await page.browser().newPage();
      await sheet.setViewport({ width: 900, height: 1200, deviceScaleFactor: 1 });
      await sheet.goto(page.url(), { waitUntil: 'domcontentloaded' });
      await sheet.setContent(built[which], { waitUntil: 'load', timeout: 60000 });
      await sheet.waitForFunction(() => document.documentElement.getAttribute('data-ws-fonts') === 'ready', { timeout: 15000 }).catch(() => {});
      await sheet.evaluate(() => document.fonts && document.fonts.ready);
      await new Promise(r => setTimeout(r, 150));
      res[which] = await sheet.evaluate(() => {
        const PX = 96 / 25.4;
        const cells = [...document.querySelectorAll('.ws-cell')];
        const out = { cells: cells.length, minFontPt: 99, minL: 99, minR: 99, minT: 99, minB: 99, ko: 0, worst: [], gridCols: 0, emptySlots: 0 };
        const ink = (el, lab) => {
          let b = null;
          const walk = (n) => {
            if (lab && (n === lab || lab.contains(n))) return;
            if (n.nodeType === 3) {
              if (!n.textContent.trim()) return;
              const r = document.createRange(); r.selectNodeContents(n);
              for (const q of r.getClientRects()) { if (q.width && q.height) b = b ? { l: Math.min(b.l, q.left), r: Math.max(b.r, q.right), t: Math.min(b.t, q.top), b: Math.max(b.b, q.bottom) } : { l: q.left, r: q.right, t: q.top, b: q.bottom }; }
              const fs = parseFloat(getComputedStyle(n.parentElement).fontSize) * 0.75;
              if (/\d/.test(n.textContent)) out.minFontPt = Math.min(out.minFontPt, fs);
              return;
            }
            if (n.nodeType !== 1) return;
            const cs = getComputedStyle(n);
            if (cs.display === 'none' || cs.visibility === 'hidden') return;
            const tag = n.tagName.toLowerCase();
            const borders = ['Top', 'Right', 'Bottom', 'Left'].some(s => parseFloat(cs['border' + s + 'Width']) > 0 && cs['border' + s + 'Style'] !== 'none');
            if (tag === 'svg' || tag === 'path' || tag === 'line' || borders) {
              const q = n.getBoundingClientRect();
              if (q.width || q.height) b = b ? { l: Math.min(b.l, q.left), r: Math.max(b.r, q.right), t: Math.min(b.t, q.top), b: Math.max(b.b, q.bottom) } : { l: q.left, r: q.right, t: q.top, b: q.bottom };
              if (tag === 'svg') return;
            }
            for (const ch of n.childNodes) walk(ch);
          };
          for (const ch of el.childNodes) walk(ch);
          return b;
        };
        for (const cell of cells) {
          const cr = cell.getBoundingClientRect();
          const lab = cell.querySelector('[data-ws-label]');
          const b = ink(cell, lab);
          if (!b) continue;
          const L = (b.l - cr.left) / PX, R = (cr.right - b.r) / PX, T = (b.t - cr.top) / PX, B = (cr.bottom - b.b) / PX;
          out.minL = Math.min(out.minL, L); out.minR = Math.min(out.minR, R); out.minT = Math.min(out.minT, T); out.minB = Math.min(out.minB, B);
          if (lab) {
            const lr = lab.getBoundingClientRect();
            const ks = Math.max(lr.right - cr.left, lr.bottom - cr.top) / PX + 1;  // keep-out square side mm
            if ((b.l - cr.left) / PX < ks && (b.t - cr.top) / PX < ks) {
              // any ink rect inside the square? approximate with bbox corner: check text rects individually
              let hit = false;
              const rects = [];
              cell.querySelectorAll('*').forEach(e => { if (lab.contains(e) || e === lab) return; for (const n of e.childNodes) if (n.nodeType === 3 && n.textContent.trim()) { const r = document.createRange(); r.selectNodeContents(n); for (const q of r.getClientRects()) rects.push(q); } });
              cell.querySelectorAll('svg').forEach(s => rects.push(s.getBoundingClientRect()));
              for (const q of rects) if ((q.left - cr.left) / PX < ks && (q.top - cr.top) / PX < ks && q.width > 0) hit = true;
              if (hit) out.ko++;
            }
          }
          if (L < 1 || R < 1) out.worst.push({ txt: cell.textContent.replace(/\s+/g, ' ').trim().slice(0, 24), L: +L.toFixed(2), R: +R.toFixed(2), wmm: +(cr.width / PX).toFixed(1) });
        }

        // r2: arc gaps (both sides) and grid cell widths
        out.arcMinR = 99; out.arcMinL = 99; out.arcN = 0; out.arcWorst = '';
        const ctxA = document.createElement('canvas').getContext('2d'); const K = 20;
        const inkOf = (sp) => { const rg = document.createRange(); rg.selectNodeContents(sp); const r = rg.getBoundingClientRect(); const cs = getComputedStyle(sp);
          ctxA.font = `${cs.fontStyle} ${cs.fontWeight} ${parseFloat(cs.fontSize) * K}px ${cs.fontFamily}`; const m = ctxA.measureText(sp.textContent.trim());
          const fa = m.fontBoundingBoxAscent / K, fd = m.fontBoundingBoxDescent / K; const base = r.top + (r.height - (fa + fd)) / 2 + fa;
          return { l: r.left - m.actualBoundingBoxLeft / K, r: r.left + m.actualBoundingBoxRight / K, t: base - m.actualBoundingBoxAscent / K, b: base + m.actualBoundingBoxDescent / K }; };
        for (const svg of document.querySelectorAll('[data-ws-ops="division"] [role="group"] svg')) {
          const p = svg.querySelector('path'); if (!p || !/Q/.test(p.getAttribute('d') || '')) continue;
          const grid = svg.closest('[role="group"]'); const kids = [...grid.children]; const ai = kids.indexOf(svg.parentElement); if (ai < 0) continue;
          const row = svg.parentElement.style.gridRow;
          const dd = kids.slice(ai + 1).find((s) => s.style.gridRow === row && /^\d$/.test(s.textContent.trim()));
          const dv = kids.slice(0, ai).reverse().find((s) => s.style.gridRow === row && /^\d$/.test(s.textContent.trim()));
          if (!dd) continue;
          const di = inkOf(dd); const vi = dv ? inkOf(dv) : null;
          const sw = (parseFloat(p.getAttribute('stroke-width')) || 1.5) * 96 / 72 / 2; const M = p.getScreenCTM(); const L = p.getTotalLength();
          let right = -Infinity, left = Infinity;
          for (let i = 0; i <= 200; i++) { const q0 = p.getPointAtLength(L * i / 200); const q = new DOMPoint(q0.x, q0.y).matrixTransform(M);
            if (q.y >= di.t - sw && q.y <= di.b + sw) right = Math.max(right, q.x + sw);
            if (vi && q.y >= vi.t - sw && q.y <= vi.b + sw) left = Math.min(left, q.x - sw); }
          const gR = (di.l - right) / PX, gL = vi ? (left - vi.r) / PX : 99; out.arcN++;
          if (gR < out.arcMinR) { out.arcMinR = gR; out.arcWorst = grid.getAttribute('aria-label') || ''; }
          if (gL < out.arcMinL) out.arcMinL = gL;
        }
        out.arcMinR = +out.arcMinR.toFixed(2); out.arcMinL = +out.arcMinL.toFixed(2);
        const ws = new Set(); for (const g of document.querySelectorAll('.ws-grid.facts')) for (const c of g.querySelectorAll(':scope > .ws-cell')) ws.add((c.getBoundingClientRect().width / PX).toFixed(1));
        out.widths = [...ws];
        out.cols = 0; { const tops = {}; for (const c of cells) { const t = Math.round(c.getBoundingClientRect().top); tops[t] = (tops[t] || 0) + 1; } out.cols = Math.max(0, ...Object.values(tops)); }
        out.worst = out.worst.slice(0, 6);
        for (const k of ['minL', 'minR', 'minT', 'minB']) out[k] = +out[k].toFixed(2);
        return out;
      });
      if (SHOTS || SHOTONLY.includes(id)) {
        const pdf = path.join(OUT, `${id}-${which}.pdf`);
        await sheet.pdf({ path: pdf, printBackground: false, preferCSSPageSize: true });
      }
      await sheet.close();
    }
    const row = { id, fits: built.fits, notes: built.notes, pages: built.pageCount, keyPages: built.keyPageCount, n: built.n, pupil: res.pupil, key: res.key, ans: built.ans };
    rows.push(row);
    console.log(id, 'n', row.n, 'pg', row.pages, 'cells', res.pupil.cells, 'font', res.pupil.minFontPt.toFixed(1), 'L', res.pupil.minL, 'R', res.pupil.minR, 'ko', res.pupil.ko, 'cols', res.pupil.cols, 'W', res.pupil.widths.join('/'), 'arc', res.pupil.arcN, res.pupil.arcMinR, res.pupil.arcMinL, 'karc', res.key.arcMinR, res.key.arcMinL, 'keyL', res.key.minL, 'keyR', res.key.minR, 'keyko', res.key.ko, res.pupil.worst.length ? JSON.stringify(res.pupil.worst.slice(0, 2)) : '');
  }
  fs.writeFileSync(path.join(OUT, 'sweep.json'), JSON.stringify({ rows, problems }, null, 1));
  console.log('problems', problems.length, JSON.stringify(problems.slice(0, 5)));
  await close();
})().catch(e => { console.error(e); process.exit(1); });
