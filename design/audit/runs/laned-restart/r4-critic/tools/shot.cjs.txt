const puppeteer = require('puppeteer');
(async () => {
  const [,, file, outPrefix] = process.argv;
  const b = await puppeteer.launch({args:['--no-sandbox']});
  const p = await b.newPage(); await p.setViewport({width:794,height:1123,deviceScaleFactor:1});
  await p.goto('file://' + file);
  await new Promise(r=>setTimeout(r,800));
  const pages = await p.$$('.ws-page');
  for (let i = 0; i < pages.length; i++) await pages[i].screenshot({ path: `${outPrefix}-${i + 1}.png` });
  console.log(pages.length);
  await b.close();
})();
